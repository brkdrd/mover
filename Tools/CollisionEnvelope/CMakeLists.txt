cmake_minimum_required(VERSION 3.20)
project(MoverCollisionEnvelope LANGUAGES CXX)
set(DEBUG_RENDERER_IN_DEBUG_AND_RELEASE OFF CACHE BOOL "Headless" FORCE)
set(PROFILER_IN_DEBUG_AND_RELEASE OFF CACHE BOOL "Headless" FORCE)
set(INTERPROCEDURAL_OPTIMIZATION OFF CACHE BOOL "Auditable numerical build" FORCE)
set(CROSS_PLATFORM_DETERMINISTIC ON CACHE BOOL "Precise arithmetic" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../Build" "${CMAKE_BINARY_DIR}/jolt")
add_executable(CollisionEnvelope main.cpp)
target_compile_features(CollisionEnvelope PRIVATE cxx_std_17)
target_link_libraries(CollisionEnvelope PRIVATE Jolt)
